━━ WINDOWS ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Title bar  🟡 🟢 🔴                    minimize / maximize (panel stays) / close
Alt + Tab   (Alt + Shift + Tab)       hold Alt: window list, switch on release
Super + ↑                             maximize; on a half: its top quarter
Super + ↓                             restore; on a half: bottom quarter; plain: minimize
Super + ← / →                         left / right half; opposite side restores;
                                        same side again: to the next screen
Super + Shift + ← / →                 move the window to the screen on that side
Super + F                             maximize on / off (app fullscreen: off)
Super + Q                             close window
━━ MOUSE ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
drag any edge or corner               resize the window
double click a title bar              maximize <-> restore
drag a title bar (either button)      move the window (a maximized one gets its size back)
  … drop at the top = maximize, left / right edge = that half of the screen,
    near a corner = that quarter, on a workspace number in a panel = move it there;
    edges stick to screen and window edges on the way
click a number in the panel           go to that workspace (every screen has its panel)
Super + left-drag                     move a window without a title bar
Super + right-drag (inside window)    resize from the nearest corner
━━ RESIZE (keyboard) ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Super + Ctrl + ← →   (or J ;)         narrower / wider
Super + Ctrl + ↑ ↓   (or L K)         shorter / taller
━━ MINIMIZE ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Super + minus                         minimize
Super + Shift + minus                 bring the last minimized window back
━━ WORKSPACES (every screen has its own) ━━━━━━━━━━━━━━━━━━━━━━━━
Super + 1 … 6  (7 … 9 while they exist)  go to workspace N on this screen
Super + Shift + 1 … 9                 send the window there
3 fingers  ← →                        previous / next workspace on this screen
━━ LAUNCH ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Super + Enter                         terminal (kitty)
Super + D     /  3 fingers ↓          app launcher
Super + Tab   /  3 fingers ↑          window switcher
Super + R                             run a command
Super + E                             file manager
Super + C                             light-year calendar
Super + Shift + S                     screenshot of an area -> clipboard + ~/Pictures/Screenshots
Print                                 screenshot of this screen -> clipboard + file
Ctrl + Shift + Esc                    system monitor (btop)
Apple button -> Favorites             favorite apps, launch at login, workspaces
4 fingers ↑ / ↓                       maximize / restore

Ctrl + Shift                          Ukrainian (again: back to the previous one)
