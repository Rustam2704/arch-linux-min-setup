cmake_minimum_required(VERSION 3.21)
project(qt_scroll_phases LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Gui GuiPrivate Widgets Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(INPUT REQUIRED IMPORTED_TARGET libevdev libudev)
add_library(qscrollphases MODULE plugin.cpp phases.cpp)
target_link_libraries(qscrollphases PRIVATE Qt6::GuiPrivate PkgConfig::INPUT)
set_target_properties(qscrollphases PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/generic")
add_executable(test_phases test_phases.cpp phases.cpp)
target_link_libraries(test_phases PRIVATE Qt6::Widgets Qt6::GuiPrivate Qt6::Test PkgConfig::INPUT)
enable_testing()
add_test(NAME phases COMMAND test_phases -platform offscreen)
add_executable(probe probe.cpp)
target_link_libraries(probe PRIVATE Qt6::Gui)
